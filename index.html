import { serve } from "https://deno.land/std@0.168.0/http/server.ts" const corsHeaders = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'POST, OPTIONS', 'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type', } serve(async (req) => { if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders }) try { const { positions, userMessage } = await req.json() const GEMINI_KEY = Deno.env.get('GEMINI_API_KEY') // Minimização extrema de dados para evitar o erro "Resource Exhausted" const leanPositions = (positions || []).slice(0, 3).map((p: any) => ({ s: p.symbol, pnl: parseFloat(p.unrealisedPnl || 0).toFixed(0), l: p.leverage + 'x' })); const systemPrompt = `BALTAZAR: Gestor de Risco de Alta Intensidade. Resiliência total. Sê frio. Usa **negritos** para valores.` // Usamos v1beta com o ID estável que a Google aceita para faturamento const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${GEMINI_KEY}` const aiRes = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: `${systemPrompt}\n\nData: ${JSON.stringify(leanPositions)}\n\nQuery: ${userMessage || "Relatório"}` }] }], generationConfig: { temperature: 0.2, maxOutputTokens: 300 } // Limita o custo e o uso de recursos }) }) const data = await aiRes.json() if (data.error) { // Se o 1.5-flash falhar, tentamos o 2.0-flash como fallback imediato const fallbackUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${GEMINI_KEY}` const fallbackRes = await fetch(fallbackUrl, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: `${systemPrompt}\n\nStatus: ${JSON.stringify(leanPositions)}` }] }] }) }); const fallbackData = await fallbackRes.json(); if (fallbackData.error) throw new Error(data.error.message); return new Response(JSON.stringify({ insight: fallbackData.candidates?.[0]?.content?.parts?.[0]?.text || "Baltazar Online." }), { headers: corsHeaders }); } const insight = data.candidates?.[0]?.content?.parts?.[0]?.text || "Em standby."; return new Response(JSON.stringify({ insight }), { headers: corsHeaders }); } catch (e) { return new Response(JSON.stringify({ insight: `**SISTEMA:** Erro de Quota/Modelo. Google ainda a processar o teu plano pago. Aguarda 60s. (${e.message})` }), { headers: corsHeaders }); } })
